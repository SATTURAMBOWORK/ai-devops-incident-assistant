# Exact versions, not ranges.
#
# scikit-learn writes model.joblib in a format tied to its own version and to
# numpy's. Training with one version and loading with another can fail outright,
# or - worse - load and silently predict differently. Pinning is what makes
# "it worked on my machine" also true inside the container and in CI.
#
# Regenerate after changing a dependency:
#   .venv/Scripts/python.exe -m pip freeze > requirements.txt

threadpoolctl==3.6.0 # pulled in by scikit-learn, pinned so builds are reproducible
